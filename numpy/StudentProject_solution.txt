34) Create grade array (A/B/C/D/F).
    90+  -> A
    80-89 -> B
    70-79 -> C
    60-69 -> D
    <60 -> F
percentage = (np.sum(marks, axis=1) / 500) * 100 
grades = np.empty(len(percentage), dtype='<U2') 
grades[percentage >= 90] = "A" 
grades[(percentage >= 80) & (percentage < 90)] = "B" 
grades[(percentage >= 70) & (percentage < 80)] = "C" 
grades[(percentage >= 60) & (percentage < 70)] = "D" 
grades[percentage < 60] = "F" 
for name, grade in zip(data[:,1], grades): 
    print(f"{name:12} : {grade}")
